GenericObjectPool.cs 
public class ObjectPool<T> where T : Component
{
    private readonly Queue<T> _poolQueue = new();
    private readonly T _prefab;
    private readonly Transform _parentRoot;

    public ObjectPool(T prefab, int initialSize, Transform parent = null)
    {
        _prefab = prefab;
        _parentRoot = parent;
        for (int i = 0; i < initialSize; i++)
            _poolQueue.Enqueue(CreateNewInstance());
    }

    public T Spawn(Vector3 position, Quaternion rotation)
    {
        T item = _poolQueue.Count > 0 ? _poolQueue.Dequeue() : CreateNewInstance();
        item.transform.SetPositionAndRotation(position, rotation);
        item.gameObject.SetActive(true);
        return item;
    }

    public void ReturnToPool(T item)
    {
        item.gameObject.SetActive(false);
        _poolQueue.Enqueue(item);
    }
}

코드 리뷰 & 트러블슈팅 노트

기존 문제점 (Before)

매 프레임 수십 개의 총알과 파티클을 Instantiate() 및 Destroy()하면서 가비지 컬렉터(GC) 호출이 잦아져 주기적인 프레임 드랍(스파이크) 현상 발생.

해결 및 개선점 (After)

  • 제네릭(where T : Component) 도입으로 타입 안전성 보장 및 캐스팅 비용 절감
  • 초기 로딩 시 큐에 프리웜(Pre-warm) 할당하여 런타임 힙 메모리 할당 0 바이트 달성
  • 프로파일러 측정 결과 60 FPS 안정 유지 성공
💡 향후 리팩토링 고려사항: Unity 2021+ 내장 UnityEngine.Pool과의 벤치마크 및 리액티브 이벤트 연동 검토.
StateMachine.cs 
public interface IState
{
    void Enter();
    void Execute();
    void Exit();
}

public class StateMachine
{
    public IState CurrentState { get; private set; }

    public void ChangeState(IState newState)
    {
        if (CurrentState == newState) return;

        CurrentState?.Exit();
        CurrentState = newState;
        CurrentState.Enter();
    }

    public void Update()
    {
        CurrentState?.Execute();
    }
}

코드 리뷰 & 트러블슈팅 노트

기존 문제점 (Before)

초기에는 거대한 switch-case 구문과 수많은 boolean 플래그로 몬스터 AI를 제어하여, 상태가 추가될 때마다 버그가 빈번하고 코드 가독성이 급격히 저하됨.

해결 및 개선점 (After)

  • 상태별 진입(Enter), 실행(Execute), 탈출(Exit) 라이프사이클 명확화
  • 단일 책임 원칙(SRP) 준수로 버그 추적 시간 60% 단축
  • 신규 보스 몬스터 패턴 추가 시 기존 코드 수정 없이 새 State 클래스만 구현하여 OCP 충족
💡 추가 고도화 아이디어: 복잡한 보스 패턴을 위한 계층형 상태 머신(HFSM) 또는 비헤이비어 트리(Behavior Tree) 도입 검토.
